"""Replay the Gate 5 hour run's real HTTP statuses through the fixed
classify_fetch_outcome(), to confirm the fix does not change that run's
already-committed evidence -- it only changes behaviour for status codes that
did not occur in it.
"""
import sys
sys.path.insert(0, "services/ingest")
import psycopg
from ingest.run import classify_fetch_outcome, is_failure

dsn = "postgresql://signalbox:local_dev_gate5_only@127.0.0.1:55432/signalbox"
conn = psycopg.connect(dsn)
cur = conn.cursor()
cur.execute("select status, error from tenant_hsl_tripupdates.fetch order by id")
rows = cur.fetchall()

print(f"rows in this run's fetch table: {len(rows)}")
outcomes = {}
for status, error in rows:
    outcome = classify_fetch_outcome(status, has_error=bool(error))
    outcomes[outcome] = outcomes.get(outcome, 0) + 1

print("outcome counts under the FIXED classifier, replayed over real data:")
for outcome, count in sorted(outcomes.items()):
    print(f"  {outcome:20} {count:5}  failure={is_failure(outcome)}")

failures = sum(c for o, c in outcomes.items() if is_failure(o))
print()
print(f"total failures under the fixed logic: {failures}")
print(f"committed report's pipeline_outcomes.failures: 0")
print(f"match: {failures == 0}")
assert failures == 0, "replay disagrees with the committed report -- investigate before proceeding"
assert "unexpected_status" not in outcomes, "this run's data actually contained an unexpected status"
print()
print("Confirmed: this run's committed evidence is numerically identical under")
print("the fixed classifier. The fix changes behaviour only for status codes")
print("this run never received.")
