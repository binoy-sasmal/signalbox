status outside 200/304/transport-error built an outcome string
(f"unexpected_{status}") that could never be a member of the closed

pipeline_outcomes.failures -- the number Gate 8's SLI 1 reads.

the suite catch it, restore, confirm clean. A no-op detector is included
because the first attempt at mutation 2 silently matched nothing -- the
harness itself needed the same discipline it exists to enforce.

#!/bin/sh
set -u
PY=.venv/Scripts/python.exe
run_suite() { "$PY" -m unittest discover -s services/ingest/tests -p 'test_*.py' 2>&1 | tail -4; }

mutate() {
  cp "$1" "$1.orig"
  before=$(md5sum "$1")
  "$PY" - "$1" <<PYEOF
import io,sys
p=sys.argv[1]; s=io.open(p,encoding='utf-8').read()
$2
io.open(p,'w',encoding='utf-8',newline='\n').write(s)
PYEOF
  after=$(md5sum "$1")
  echo "### MUTATION: $3"
  if [ "$before" = "$after" ]; then
    echo "!!! NO-OP: the replace() pattern did not match anything. Fix the harness. !!!"
  fi
  echo "--- suite result ---"
  run_suite
  mv "$1.orig" "$1"
  echo
}

echo "### control: unmutated suite ###"
run_suite
echo

mutate services/ingest/ingest/run.py \
"s=s.replace('    return \"unexpected_status\"', '    return f\"unexpected_{status}\"')" \
"reproduce the original bug -- a fresh string per status code again"

mutate services/ingest/ingest/run.py \
"s=s.replace('    \"unexpected_status\",\n', '')" \
"remove unexpected_status from FAILURE_OUTCOMES's declared set (both places it appears -- OUTCOMES and FAILURE_OUTCOMES)"

mutate services/ingest/ingest/run.py \
"s=s.replace('            + counters.unexpected_statuses + counters.write_failures', '            + counters.write_failures')" \
"stop adding unexpected_statuses into the report's failures sum, even though the taxonomy and counter both still exist"

echo "### final control: tree restored ###"
run_suite
