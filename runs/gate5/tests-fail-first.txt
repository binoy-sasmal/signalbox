#!/bin/sh
# Prove each new gate can fail, by breaking the thing it guards.
# A test that only ever passes and a test that cannot fail look identical from
# outside (review log row 5).
set -u
PY=.venv/Scripts/python.exe
run_suite() { "$PY" -m unittest discover -s services/ingest/tests -p 'test_*.py' 2>&1 | tail -4; }

mutate() {  # $1=file $2=python-replacement-script $3=label
  cp "$1" "$1.orig"
  "$PY" - "$1" <<PYEOF
import io,sys
p=sys.argv[1]; s=io.open(p,encoding='utf-8').read()
$2
io.open(p,'w',encoding='utf-8',newline='\n').write(s)
PYEOF
  echo "### MUTATION: $3"
  echo "--- suite result ---"
  run_suite
  mv "$1.orig" "$1"
  echo
}

echo "### control: unmutated suite ###"
run_suite
echo

mutate services/ingest/ingest/decode.py \
"s=s.replace('    if trip.trip_id:\n        return f\"trip:{trip.trip_id}@{trip.start_date}\", \"trip_id\"\n', '    if True:\n        return f\"trip:{trip.trip_id}@{trip.start_date}\", \"trip_id\"\n')" \
"revert the dedup key to (trip_id, start_date) -- the bug this gate found"

mutate services/ingest/ingest/decode.py \
"s=s.replace('    if ratio < tolerance:', '    if False:')" \
"disable the key-collapse guard entirely"

mutate services/ingest/ingest/dropqueue.py \
"s=s.replace('evicted = self._items.popleft()', 'evicted = self._items.pop()')" \
"invert the drop policy to drop-NEWEST"

mutate services/ingest/ingest/dropqueue.py \
"s=s.replace('                self.dropped += 1\n', '')" \
"stop counting drops"

mutate services/ingest/ingest/run.py \
"s=s.replace('FAILURE_OUTCOMES = frozenset({\n    \"dropped\",', 'FAILURE_OUTCOMES = frozenset({')" \
"remove 'dropped' from the failure set -- the error-budget laundering ADR 0010 exists to stop"

mutate services/ingest/ingest/config.py \
"s=s.replace('    if raw[\"incrementality\"] != SUPPORTED_INCREMENTALITY:', '    if False:')" \
"stop refusing a DIFFERENTIAL feed"

mutate services/ingest/ingest/poller.py \
"s=s.replace('        while target <= now:', '        while False:')" \
"stop skipping missed ticks"

echo "### final control: tree restored ###"
run_suite
